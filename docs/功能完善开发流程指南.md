# 蜜罐诱捕系统功能完善开发流程指南

## 1. 概述

### 1.1 目的
本指南针对Andorralee v2.0蜜罐诱捕系统在测试过程中发现的功能缺失和完善需求，提供系统性的开发流程指南，以确保系统完全符合《测试大纲》要求。

### 1.2 适用范围
- 前端可视化界面开发
- 高级攻击溯源功能增强
- 系统集成和优化
- 性能调优和安全加固

### 1.3 开发优先级
根据测试结果和用户需求，按以下优先级进行开发：
1. **优先级1**: 前端可视化界面开发 (测试项6缺失)
2. **优先级2**: 高级攻击溯源功能增强 (测试项3部分完成)
3. **优先级3**: 系统性能优化和安全加固

       if (chartRef.current) {
         const chart = echarts.init(chartRef.current);
         const chartOptions = {
           xAxis: { type: 'category', data: data.map(d => d.time) },
           yAxis: { type: 'value' },
           series: [{ data: data.map(d => d.value), type: 'line' }],
           ...options
         };
         chart.setOption(chartOptions);
         
         return () => chart.dispose();
       }
     }, [data, options]);
     
     return <div ref={chartRef} style={{ width: '100%', height: '300px' }} />;
   };

#### 3.2.1 攻击链分析算法
```go
// internal/services/attack_analysis_service.go
type AttackChainAnalyzer struct {
    sessionRepo repositories.SessionRepository
    logRepo     repositories.LogRepository
}

type AttackChain struct {
    AttackerIP    string                 `json:"attacker_ip"`
    StartTime     time.Time             `json:"start_time"`
    EndTime       time.Time             `json:"end_time"`
    AttackStages  []AttackStage         `json:"attack_stages"`
    ThreatLevel   string                `json:"threat_level"`
    Confidence    float64               `json:"confidence"`
}

type AttackStage struct {
    Stage       string    `json:"stage"`       // reconnaissance, exploitation, persistence
    Timestamp   time.Time `json:"timestamp"`
    Actions     []string  `json:"actions"`
    Evidence    []string  `json:"evidence"`
}

func (a *AttackChainAnalyzer) AnalyzeAttackChain(attackerIP string, timeRange TimeRange) (*AttackChain, error) {
    // 1. 获取攻击者的所有会话
    sessions, err := a.sessionRepo.GetSessionsByIP(attackerIP, timeRange)
    if err != nil {
        return nil, err
    }
    
    // 2. 获取相关日志
    logs, err := a.logRepo.GetLogsByIP(attackerIP, timeRange)
    if err != nil {
        return nil, err
    }
    
    // 3. 分析攻击阶段
    stages := a.identifyAttackStages(sessions, logs)
    
    // 4. 计算威胁等级
    threatLevel := a.calculateThreatLevel(stages)
    
    // 5. 计算置信度
    confidence := a.calculateConfidence(stages)
    
    return &AttackChain{
        AttackerIP:   attackerIP,
        StartTime:    sessions[0].StartTime,
        EndTime:      sessions[len(sessions)-1].EndTime,
        AttackStages: stages,
        ThreatLevel:  threatLevel,
        Confidence:   confidence,
    }, nil
}

func (a *AttackChainAnalyzer) identifyAttackStages(sessions []Session, logs []Log) []AttackStage {
    var stages []AttackStage
    
    // 侦察阶段识别
    if a.hasReconnaissancePatterns(logs) {
        stages = append(stages, AttackStage{
            Stage:     "reconnaissance",
            Timestamp: a.getEarliestReconTime(logs),
            Actions:   a.extractReconActions(logs),
            Evidence:  a.extractReconEvidence(logs),
        })
    }
    
    // 利用阶段识别
    if a.hasExploitationPatterns(logs) {
        stages = append(stages, AttackStage{
            Stage:     "exploitation",
            Timestamp: a.getEarliestExploitTime(logs),
            Actions:   a.extractExploitActions(logs),
            Evidence:  a.extractExploitEvidence(logs),
        })
    }
    
    // 权限维持阶段识别
    if a.hasPersistencePatterns(logs) {
        stages = append(stages, AttackStage{
            Stage:     "persistence",
            Timestamp: a.getEarliestPersistenceTime(logs),
            Actions:   a.extractPersistenceActions(logs),
            Evidence:  a.extractPersistenceEvidence(logs),
        })
    }
    
    return stages
}
```

// internal/services/behavior_profiling_service.go
type BehaviorProfiler struct {
    sessionRepo repositories.SessionRepository
    commandRepo repositories.CommandRepository
}

type AttackerProfile struct {
    AttackerIP        string            `json:"attacker_ip"`
    ProfiledAt        time.Time         `json:"profiled_at"`
    SkillLevel        string            `json:"skill_level"`        // novice, intermediate, advanced
    AttackStyle       string            `json:"attack_style"`       // automated, manual, hybrid
    PreferredTargets  []string          `json:"preferred_targets"`  // services, protocols
    CommandPatterns   []CommandPattern  `json:"command_patterns"`
    ThreatScore       float64           `json:"threat_score"`
    Indicators        []string          `json:"indicators"`
}

type CommandPattern struct {
    Command   string  `json:"command"`
    Frequency int     `json:"frequency"`
    Context   string  `json:"context"`
}

func (b *BehaviorProfiler) ProfileAttacker(attackerIP string) (*AttackerProfile, error) {
    // 1. 获取攻击者的历史行为数据
    sessions, err := b.sessionRepo.GetAllSessionsByIP(attackerIP)
    if err != nil {
        return nil, err
    }
    
    commands, err := b.commandRepo.GetCommandsByIP(attackerIP)
    if err != nil {
        return nil, err
    }
    
    // 2. 分析技能水平
    skillLevel := b.analyzeSkillLevel(commands)
    
    // 3. 识别攻击风格
    attackStyle := b.identifyAttackStyle(sessions, commands)
    
    // 4. 提取命令模式
    commandPatterns := b.extractCommandPatterns(commands)
    
    // 5. 计算威胁分数
    threatScore := b.calculateThreatScore(sessions, commands)
    
    // 6. 生成威胁指标
    indicators := b.generateThreatIndicators(sessions, commands)
    
    return &AttackerProfile{
        AttackerIP:       attackerIP,
        ProfiledAt:       time.Now(),
        SkillLevel:       skillLevel,
        AttackStyle:      attackStyle,
        CommandPatterns:  commandPatterns,
        ThreatScore:      threatScore,
        Indicators:       indicators,
    }, nil
}

func (c *CacheService) CacheAttackStats(key string, data interface{}, ttl time.Duration) error {
    jsonData, err := json.Marshal(data)
    if err != nil {
        return err
    }
    
    return c.redis.Set(context.Background(), key, jsonData, ttl).Err()
}

func (c *CacheService) GetCachedAttackStats(key string) (interface{}, error) {
    val, err := c.redis.Get(context.Background(), key).Result()

    return data, err
}

// utils/encryption.go
func EncryptSensitiveData(data string) (string, error) {
    key := getEncryptionKey()
    block, err := aes.NewCipher(key)
    if err != nil {
        return "", err
    }
    
    gcm, err := cipher.NewGCM(block)
    if err != nil {
        return "", err
    }
    
    nonce := make([]byte, gcm.NonceSize())
    if _, err = io.ReadFull(rand.Reader, nonce); err != nil {
        return "", err
    }
    
    ciphertext := gcm.Seal(nonce, nonce, []byte(data), nil)
    return base64.StdEncoding.EncodeToString(ciphertext), nil
}
